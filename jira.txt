from sentence_transformers import SentenceTransformer
from pyspark.sql import Row
from datetime import datetime
import uuid

# Sample data
records = [
    {
        "source": "Jira",
        "type": "Story",
        "key": "STORY-001",
        "summary": "Allow user to reset password",
        "description": "As a user, I want a reset password option on the login page.",
        "severity": "Medium",
        "priority": "High",
        "created_date": datetime(2024, 6, 1),
        "closed_date": datetime(2024, 6, 10)
    },
    {
        "source": "Jira",
        "type": "Defect",
        "key": "DEFECT-001",
        "summary": "Password reset throws error",
        "description": "Clicking reset password causes a NullPointerException in logs.",
        "severity": "High",
        "priority": "High",
        "created_date": datetime(2024, 6, 11),
        "closed_date": datetime(2024, 6, 12)
    },
    {
        "source": "ServiceNow",
        "type": "Change",
        "key": "CHG-001",
        "summary": "Database schema update for accounts",
        "description": "Add new columns for customer preferences",
        "severity": "Low",
        "priority": "Medium",
        "created_date": datetime(2024, 5, 20),
        "closed_date": datetime(2024, 5, 25)
    }
]

# Load model
model = SentenceTransformer('all-MiniLM-L6-v2')

# Prepare Spark Rows with embedding
data = []
for record in records:
    full_text = f"{record['summary']}. {record['description']}. Severity: {record['severity']}. Priority: {record['priority']}"
    embedding = model.encode([full_text])[0].tolist()
    row = Row(
        source=record["source"],
        type=record["type"],
        key=record["key"],
        id=str(uuid.uuid4()),
        summary=record["summary"],
        description=record["description"],
        severity=record["severity"],
        priority=record["priority"],
        created_date=record["created_date"],
        closed_date=record["closed_date"],
        embedding=embedding
    )
    data.append(row)

df = spark.createDataFrame(data)

# Save to Delta table on blob container
df.write.format("delta").mode("overwrite").save("/mnt/vector_data/vector_db")
