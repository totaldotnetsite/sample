import faiss
import numpy as np

# Load from Delta
df = spark.read.table("your_catalog.vector_db").toPandas()
embeddings = np.vstack(df["embedding"].values)
index = faiss.IndexFlatL2(embeddings.shape[1])  # Euclidean distance index
index.add(embeddings)

# Search with a query
from sentence_transformers import SentenceTransformer
model = SentenceTransformer('all-MiniLM-L6-v2')
query = "Error resetting password"
query_vec = model.encode([query]).astype("float32")

D, I = index.search(query_vec, k=3)  # top 3 results

# Show matching rows
results = df.iloc[I[0]]
print(results[["key", "summary", "type"]])
