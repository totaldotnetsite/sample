# Required libraries
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity
import pandas as pd
import numpy as np
import requests
from typing import List

# Step 1: Load embedding model
embedding_model = SentenceTransformer('all-MiniLM-L6-v2')

# Step 2: Load data (assume stories_df and defects_df are preloaded)
# stories_df: story_id, title, description, acceptance_criteria, comments, closed_date
# defects_df: defect_id, title, description, comments, created_date

# Step 3: Combine text for embedding
def combine_story_text(row):
    return f"ID: {row['story_id']}\nTitle: {row['title']}\nDescription: {row['description']}\nAcceptance Criteria: {row['acceptance_criteria']}\nComments: {row['comments']}"

def combine_defect_text(row):
    return f"Defect ID: {row['defect_id']}\nTitle: {row['title']}\nDescription: {row['description']}\nComments: {row['comments']}"

stories_df['combined_text'] = stories_df.apply(combine_story_text, axis=1)
defects_df['combined_text'] = defects_df.apply(combine_defect_text, axis=1)

# Step 4: Create story embeddings
story_embeddings = embedding_model.encode(stories_df['combined_text'].tolist(), convert_to_tensor=True)

# Step 5: Define function to query LLaMA 4 Maverick endpoint
def query_llama4(prompt: str, endpoint_url: str, api_token: str) -> str:
    headers = {
        "Authorization": f"Bearer {api_token}",
        "Content-Type": "application/json"
    }
    data = {"inputs": prompt}
    response = requests.post(endpoint_url, headers=headers, json=data)
    response.raise_for_status()
    return response.json()["predictions"][0]["text"]

# Step 6: Process each defect
defect_links = []
ENDPOINT_URL = "https://<your-databricks-workspace>/serving-endpoints/<your-endpoint>/invocations"
API_TOKEN = "<YOUR_DATABRICKS_TOKEN>"

for _, defect in defects_df.iterrows():
    defect_created = defect['created_date']
    filtered_stories = stories_df[stories_df['closed_date'] <= defect_created]
    if filtered_stories.empty:
        continue

    filtered_embeddings = embedding_model.encode(filtered_stories['combined_text'].tolist(), convert_to_tensor=True)
    defect_embedding = embedding_model.encode(defect['combined_text'], convert_to_tensor=True)

    sims = cosine_similarity([defect_embedding.cpu().numpy()], filtered_embeddings.cpu().numpy())[0]
    top_indices = np.argsort(sims)[::-1][:5]  # Top 5 most similar stories
    top_stories = filtered_stories.iloc[top_indices]

    # Construct prompt
    candidate_section = ""
    for i, row in top_stories.iterrows():
        candidate_section += f"{row['combined_text']}\n\n"

    prompt = f"""
You are a software quality expert.

A defect occurred with the following details:
{defect['combined_text']}

The following are potential causing stories:
{candidate_section}

Which of these stories most likely introduced the defect? List the story IDs and explain your reasoning.
"""

    reasoning = query_llama4(prompt, ENDPOINT_URL, API_TOKEN)
    linked_story_ids = top_stories['story_id'].tolist()

    defect_links.append({
        "defect_id": defect['defect_id'],
        "linked_story_ids": linked_story_ids,
        "reasoning": reasoning.strip()
    })

# Step 7: Output result
linked_df = pd.DataFrame(defect_links)
linked_df.to_json("linked_defects_stories.json", orient="records", indent=2)
