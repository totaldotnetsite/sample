from pyspark.sql.types import *
from pyspark.sql import Row
import uuid
from datetime import datetime
from sentence_transformers import SentenceTransformer

# Load embedding model
model = SentenceTransformer("all-MiniLM-L6-v2")

# Sample record
records = [
    {
        "source": "Jira",
        "type": "Defect",
        "key": "DEF-001",
        "summary": "Password reset throws error",
        "description": "Clicking reset password causes a NullPointerException.",
        "severity": "High",
        "priority": "High",
        "created_date": datetime(2024, 6, 11),
        "closed_date": datetime(2024, 6, 12)
    }
]

# Generate embedding
rows = []
for record in records:
    text = f"{record['summary']}. {record['description']}. Severity: {record['severity']}. Priority: {record['priority']}"
    embedding = model.encode([text])[0].tolist()
    rows.append((
        record["source"],
        record["type"],
        record["key"],
        str(uuid.uuid4()),
        record["summary"],
        record["description"],
        record["severity"],
        record["priority"],
        record["created_date"],
        record["closed_date"],
        embedding
    ))

# Define schema
schema = StructType([
    StructField("source", StringType()),
    StructField("type", StringType()),
    StructField("key", StringType()),
    StructField("id", StringType()),
    StructField("summary", StringType()),
    StructField("description", StringType()),
    StructField("severity", StringType()),
    StructField("priority", StringType()),
    StructField("created_date", TimestampType()),
    StructField("closed_date", TimestampType()),
    StructField("embedding", ArrayType(FloatType()))   # IMPORTANT
])

# Create DataFrame with schema
df = spark.createDataFrame(rows, schema=schema)

# Write to Delta
df.write.format("delta").mode("overwrite").save("/mnt/vector_data/vector_db")
